<script>
    function switchScreen(screen) {
        // Hide all screens
        document.getElementById('screen-viewer').classList.add('hidden');
        document.getElementById('screen-selection').classList.add('hidden');
        document.getElementById('screen-search').classList.add('hidden');

        // Update buttons
        document.querySelectorAll('.demo-btn').forEach(b => b.classList.remove('active'));

        // Show selected screen
        if(screen === 'viewer') {
            document.getElementById('screen-viewer').classList.remove('hidden');
            document.querySelector('.demo-btn:nth-child(1)').classList.add('active');
            // Ensure details panel is visible in this mode as per requirement "viewer with details"
            document.getElementById('details-panel').style.transform = 'translateX(0)';
        } else if (screen === 'selection') {
            document.getElementById('screen-selection').classList.remove('hidden');
            document.querySelector('.demo-btn:nth-child(2)').classList.add('active');
        } else if (screen === 'search') {
            document.getElementById('screen-search').classList.remove('hidden');
            document.querySelector('.demo-btn:nth-child(3)').classList.add('active');
        }
    }
    
    // Initialize
    switchScreen('viewer');
</script>
